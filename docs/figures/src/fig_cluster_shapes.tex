% cluster_shapes.png: the eight cluster shapes of the energy model, with the number of such clusters per site of an infinite fcc crystal.
% Make it in this folder with, for any scratch folder T:
%   blender -b -P render_clusters.py -- T/raw
%   python3 prepare_renderings.py T/raw T/img
%   TEXINPUTS=.:T/img: pdflatex -output-directory=T fig_cluster_shapes.tex
%   python3 pdf_to_png_dithered.py T/fig_cluster_shapes.pdf ../cluster_shapes.png
\documentclass[11pt,tikz,border=1.5pt]{standalone}
\input{figstyle}
\begin{document}
\begin{tikzpicture}
  \foreach \file/\name/\num [count=\i from 0] in {site/single site/1, pair1/pair 1/6, pair2/pair 2/3, pair3/pair 3/12} {
    \node[img] (c\i) at (4.1*\i,0) {\includegraphics[width=3.25cm]{cluster_\file}};
    \node[capt] at (c\i.south) {\name\\[-1pt] \num\ per site};
  }
  \foreach \file/\name/\num [count=\i from 0] in {triplet111/triplet 1-1-1/8, triplet112/triplet 1-1-2/12,
      triplet113/triplet 1-1-3/24, triplet123/triplet 1-2-3/24} {
    \node[img] (d\i) at (4.1*\i,-4.70) {\includegraphics[width=3.25cm]{cluster_\file}};
    \node[capt] at (d\i.south) {\name\\[-1pt] \num\ per site};
  }
  \foreach \x/\c/\t in {1.45/cOne/{1: first neighbors}, 6.0/cTwo/{2: second neighbors}, 11.0/cThree/{3: third neighbors}} {
    \draw[\c, line width=3.6pt] (\x,-6.35) -- ++(0.6,0);
    \node[leg] at ({\x+0.75},-6.35) {\t};
  }
\end{tikzpicture}
\end{document}
